@functions {
    public string GetStatusColor(string status)
    {
        if (string.IsNullOrEmpty(status)) return "gray";

        switch (status.ToLower())
        {
            case "paid": return "#007bff";
            case "approved": return "green"; // Bootstrap Blue
            case "rejected": return "red";
            case "unpaid": return "orange";
            default: return "gray";
        }
    }
}

$(document).on('click', '.send-request-btn', function () {
    var btn = $(this);
    var budgetId = btn.data("id");

    // Show spinner, change text
    btn.find('.spinner-border').removeClass('d-none');
    btn.find('.btn-text').text("Sending...");

        data: { id: budgetId },
        success: function (response) {
            if (response.success) {
                iziToast.success({
                    title: 'Success',
                    message: response.message,
                    position: 'topCenter'
                });

                var row = btn.closest('tr');

                // ✅ Update Status column (index 10)
                row.find('td').eq(10).html('<span style="color:#007bff;">Approved</span>');


                // ✅ Update Send Request button to "Request Sent"
                btn.find('.spinner-border').addClass('d-none');
                btn.find('.btn-text').html('<i class="fa fa-check-circle"></i> Request Sent');
                btn.prop("disabled", true); // Disable button

                // ✅ Read ActualCost (column 7)
                var actualCostText = row.find('td').eq(7).text();
                var actualCost = parseFloat(actualCostText.replace(/,/g, ''));

                // ✅ If ActualCost > 0, add Pay button (column 8)
                if (!isNaN(actualCost) && actualCost > 0) {
                    row.find('td').eq(8).html(`
                        <a href="/AccountManager/Pay?id=${budgetId}" class="btnSS"> Pay
                        </a>
                    `);
                }

            } else {
                iziToast.error({
                    title: 'Error',
                    message: response.message,
                    position: 'topCenter'
                });

                btn.find('.spinner-border').addClass('d-none');
                btn.find('.btn-text').html('Send Request');
            }
        },
        error: function () {
            iziToast.error({
                title: 'Error',
                message: 'Something went wrong. Please try again.',
                position: 'topCenter'
            });

            btn.find('.spinner-border').addClass('d-none');
            btn.find('.btn-text').html('Send Request');
        }
    });
});
